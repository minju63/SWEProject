var checkPopupInterval;
function openChat() {
 var chat_popup_w = 460; //팝업창의 가로크기
 var chat_popup_h = 780; //팝업창의 세로크기
 if (chatWindow === undefined || chatWindow.closed) {
 var chat_popup_x = screen.availWidth - chat_popup_w; //팝업창의 가로위치
 var chat_popup_y = screen.availHeight - chat_popup_h; //팝업창의 세로위치 (상단에 띄우려면 0)
 var url = "https://jnjob.hgit.info/wise/chatbot.do";
 chatWindow = window.open("", "openForm", 'width=' + chat_popup_w + 'px, height=' + chat_popup_h + 'px, left=' + chat_popup_x + ', top=' + chat_popup_y + ', status=no, location=no, toolbar=no, menubar=no, scrollbars=no, resizable=yes');
 var myForm = document.openForm;
 myForm.action = url;
 myForm.method = "post";
 myForm.target = "openForm";
 myForm.submit();
 checkPopupInterval = setInterval(function() {
 if (chatWindow.closed) {
 clearInterval(checkPopupInterval);
 chatWindow = undefined;
 }
 }, 1000);
 } else {
 chatWindow.focus();
 }
